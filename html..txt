        function initDatabase() {
            // Load Coordinators
            const savedCoordinators = localStorage.getItem('mbg_denggungan_coordinators');
            if (savedCoordinators) {
                try {
                    coordinators = JSON.parse(savedCoordinators);
                } catch(e) {
                    coordinators = [...DEFAULT_COORDINATORS];
                }
            } else {
                coordinators = [...DEFAULT_COORDINATORS];
                saveCoordinators();
            }

            // Load Attendance Logs
            const savedLogs = localStorage.getItem('mbg_denggungan_attendance');
            if (savedLogs) {
                try {
                    attendanceLogs = JSON.parse(savedLogs);
                } catch(e) {
                    attendanceLogs = [];
                }
            } else {
                attendanceLogs = [];
                saveAttendanceLogs();
            }
        }

        function switchTab(tabName) {
            // Hide all tab contents
            document.querySelectorAll('.tab-content').forEach(el => el.classList.add('hidden'));
            
            // Remove active styling from nav buttons
            const navButtons = document.querySelectorAll('nav button');
            navButtons.forEach(btn => {
                btn.classList.remove('text-emerald-700', 'bg-emerald-50', 'shadow-sm');
                btn.classList.add('text-slate-500');
            });

            // Show selected tab content
            const activeSec = document.getElementById(`sec-${tabName}`);
            if (activeSec) activeSec.classList.remove('hidden');

            // Highlight nav button
            const activeTabBtn = document.getElementById(`tab-${tabName}`);
            if (activeTabBtn) {
                activeTabBtn.classList.remove('text-slate-500');
                activeTabBtn.classList.add('text-emerald-700', 'bg-emerald-50', 'shadow-sm');
            }

            // Stop scanner if leaving scan tab
            if (tabName !== 'scan' && html5QrcodeScanner) {
                stopScanner();
            }

            // Tab specific renders
            if (tabName === 'qrcodes') renderQRCards();
            if (tabName === 'rekap') renderRekapTable();
            if (tabName === 'data') renderCoordinatorsList();
        }

        function playSuccessBeep() {
            try {
                const ctx = new (window.AudioContext || window.webkitAudioContext)();
                const osc = ctx.createOscillator();
                const gain = ctx.createGain();
                osc.type = 'sine';
                osc.frequency.setValueAtTime(880, ctx.currentTime); // A5 note
                osc.frequency.exponentialRampToValueAtTime(1200, ctx.currentTime + 0.15);
                gain.gain.setValueAtTime(0.3, ctx.currentTime);
                gain.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + 0.2);
                osc.connect(gain);
                gain.connect(ctx.destination);
                osc.start();
                osc.stop(ctx.currentTime + 0.2);
            } catch(e) {
                console.log('Audio feedback not supported');
            }
        }

        function startScanner() {
            document.getElementById('scanner-placeholder').classList.add('hidden');
            document.getElementById('btn-start-cam').classList.add('hidden');
            document.getElementById('btn-stop-cam').classList.remove('hidden');
            document.getElementById('btn-stop-cam').classList.add('flex');

            html5QrcodeScanner = new Html5Qrcode("reader");
            
            const config = { 
                fps: 10, 
                qrbox: { width: 220, height: 220 },
                aspectRatio: 1.0
            };

            html5QrcodeScanner.start(
                { facingMode: "environment" }, 
                config, 
                onScanSuccess, 
                onScanFailure
            ).catch(err => {
                alert("Gagal membuka kamera: " + err + ". Pastikan telah memberikan izin akses kamera.");
                stopScanner();
            });
        }

        function stopScanner() {
            if (html5QrcodeScanner) {
                html5QrcodeScanner.stop().then(() => {
                    html5QrcodeScanner.clear();
                    html5QrcodeScanner = null;
                    document.getElementById('scanner-placeholder').classList.remove('hidden');
                    document.getElementById('btn-start-cam').classList.remove('hidden');
                    document.getElementById('btn-stop-cam').classList.add('hidden');
                    document.getElementById('btn-stop-cam').classList.remove('flex');
                }).catch(err => console.error(err));
            }
        }

        function setScanMode(mode) {
            scanMode = mode;
            const btnAuto = document.getElementById('btn-mode-auto');
            const btnMasuk = document.getElementById('btn-mode-masuk');
            const btnPulang = document.getElementById('btn-mode-pulang');

            [btnAuto, btnMasuk, btnPulang].forEach(btn => {

            if (mode === 'auto') {

            } else if (mode === 'masuk') {
                btnMasuk.className = "px-3 py-1 rounded-lg text-xs font-bold bg-emerald-600 text-white shadow-sm transition";
            } else if (mode === 'pulang') {
                btnPulang.className = "px-3 py-1 rounded-lg text-xs font-bold bg-amber-600 text-white shadow-sm transition";
            }
        }

        function onScanSuccess(decodedText) {
            const now = Date.now();
            // Prevent duplicate triggers within 3 seconds
            if (decodedText === lastScannedCode && (now - lastScanTime) < 3000) {
                return;
            }
            lastScannedCode = decodedText;
            lastScanTime = now;

            // Find coordinator matching the QR ID or Name
            const coord = coordinators.find(c => c.id === decodedText || c.name.toUpperCase() === decodedText.toUpperCase());

            if (!coord) {
                alert(`Kode QR tidak terdaftar! (${decodedText})`);
                return;
            }

            processAttendanceRecord(coord);
        }

        function onScanFailure(error) {
            // Quiet scan loop
        }

        function processAttendanceRecord(coord) {
            const todayStr = getTodayDateString();
            const currentTimeStr = getCurrentTimeString();

            // Find existing record for today
            let existingLog = attendanceLogs.find(log => log.coordinatorId === coord.id && log.date === todayStr);

            let recordType = 'masuk'; // default

            if (scanMode === 'auto') {
                if (!existingLog) {
                    recordType = 'masuk';
                } else if (existingLog && !existingLog.checkOut) {
                    recordType = 'pulang';
                } else {
                    alert(`${coord.name} sudah melakukan absensi Masuk (${existingLog.checkIn}) dan Pulang (${existingLog.checkOut}) hari ini!`);
                    return;
                }
            } else {
                recordType = scanMode;
            }

            playSuccessBeep();

            if (recordType === 'masuk') {
                if (existingLog) {
                    existingLog.checkIn = currentTimeStr;
                } else {
                    attendanceLogs.unshift({
                        id: 'LOG-' + Date.now(),
                        coordinatorId: coord.id,
                        name: coord.name,
                        date: todayStr,
                        checkIn: currentTimeStr,
                        checkOut: null,
                        status: 'Hadir'
                    });
                }
                showLastScanCard(coord.name, 'KEDATANGAN', currentTimeStr, 'emerald');
            } else if (recordType === 'pulang') {
                if (!existingLog) {
                    // Create entry with checkOut directly
                    attendanceLogs.unshift({
                        id: 'LOG-' + Date.now(),
                        coordinatorId: coord.id,
                        name: coord.name,
                        date: todayStr,
                        checkIn: '-',
                        checkOut: currentTimeStr,
                        status: 'Pulang'
                    });
                } else {
                    existingLog.checkOut = currentTimeStr;
                    existingLog.status = 'Lengkap';
                }
                showLastScanCard(coord.name, 'KEPULANGAN', currentTimeStr, 'amber');
            }

            saveAttendanceLogs();
            renderTodayList();
        }

        function downloadQR(coordId, coordName) {
            const container = document.getElementById(`qr-code-box-${coordId}`);
            if (!container) return;

            const img = container.querySelector('img');
            const canvas = container.querySelector('canvas');

            let imageSrc = '';
            if (img && img.src) {
                imageSrc = img.src;
            } else if (canvas) {
                imageSrc = canvas.toDataURL("image/png");
            }

            if (imageSrc) {
                const link = document.createElement('a');
                link.href = imageSrc;
                link.download = `QR_ABSENSI_${coordName.replace(/\s+/g, '_')}.png`;
                document.body.appendChild(link);
                link.click();
                document.body.removeChild(link);
            } else {
                alert('Gagal mengunduh QR Code. Silakan coba lagi.');
            }
        }

        function downloadAllQRs() {

                }, idx * 300);

        }

        function exportToExcel() {
            if (attendanceLogs.length === 0) {
                alert('Belum ada data absensi untuk diekspor!');
                return;
            }

            const filterDate = document.getElementById('filter-date').value;
            const filterMonth = document.getElementById('filter-month').value;

            let logsToExport = [...attendanceLogs];
            if (filterDate) {
                logsToExport = logsToExport.filter(l => l.date === filterDate);
            } else if (filterMonth) {
                logsToExport = logsToExport.filter(l => l.date.startsWith(filterMonth));
            }

            // Prepare Worksheet Data
            const excelData = logsToExport.map((log, index) => ({
                "No": index + 1,
                "ID Koordinator": log.coordinatorId,
                "Nama Koordinator": log.name,
                "Tanggal": log.date,
                "Jam Kedatangan": log.checkIn || '-',
                "Jam Kepulangan": log.checkOut || '-',
                "Status Absensi": log.checkOut ? 'Hadir Lengkap' : 'Belum Pulang'
            }));

            const worksheet = XLSX.utils.json_to_sheet(excelData);
            const workbook = XLSX.utils.book_new();
            XLSX.utils.book_append_sheet(workbook, worksheet, "Rekap Absensi MBG");

            // Auto-fit column widths
            const max_width = excelData.reduce((w, r) => Math.max(w, r["Nama Koordinator"].length), 10);
            worksheet["!cols"] = [
                { wch: 5 },  // No
                { wch: 15 }, // ID
                { wch: Math.max(max_width, 25) }, // Nama
                { wch: 12 }, // Tanggal
                { wch: 15 }, // Masuk
                { wch: 15 }, // Pulang
                { wch: 15 }  // Status
            ];

            const filename = `REKAP_ABSENSI_MBG_DENGGUNGAN_${filterMonth || filterDate || 'SEMUA'}.xlsx`;
            XLSX.writeFile(workbook, filename);
        }

        function getTodayDateString() {
            const today = new Date();
            const year = today.getFullYear();
            const month = String(today.getMonth() + 1).padStart(2, '0');
            const day = String(today.getDate()).padStart(2, '0');
            return `${year}-${month}-${day}`;
        }

        function getCurrentTimeString() {
            const now = new Date();
            return now.toTimeString().split(' ')[0];
        }

        }
